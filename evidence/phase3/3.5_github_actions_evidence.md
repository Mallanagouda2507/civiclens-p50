## 2. GitHub Actions Workflow Configuration (`.github/workflows/ci.yml`)

```yaml
name: CivicLens CI Pipeline

on:
  push:
    branches: [ master, main ]
  pull_request:
    branches: [ master, main ]

jobs:
  backend-test:
    name: Backend Lint, Test & Coverage
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend

    steps:
    - name: Checkout Code Repository
      uses: actions/checkout@v3

    - name: Set up Node.js 20.x Environment
      uses: actions/setup-node@v3
      with:
        node-version: 20.x
        cache: 'npm'
        cache-dependency-path: backend/package-lock.json

    - name: Install Locked Dependencies
      run: npm ci

    - name: Execute Full Jest Test Suite with Coverage
      run: npm run test:coverage
      env:
        CI: true
        NODE_ENV: test
        JWT_SECRET: ci_test_secret_key_civiclens_p50
        DISTANCE_THRESHOLD_METERS: 75
        SIMILARITY_THRESHOLD: 0.60

    - name: Upload Test Coverage Artifact
      uses: actions/upload-artifact@v3
      with:
        name: backend-coverage-report
        path: backend/coverage/

  frontend-build:
    name: Frontend Lint & Production Bundle Build
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend

    steps:
    - name: Checkout Code Repository
      uses: actions/checkout@v3

    - name: Set up Node.js 20.x Environment
      uses: actions/setup-node@v3
      with:
        node-version: 20.x
        cache: 'npm'
        cache-dependency-path: frontend/package-lock.json

    - name: Install Locked Dependencies
      run: npm ci

    - name: Validate React Build & Asset Optimization
      run: npm run build
```
